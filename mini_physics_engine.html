class PhysicsEngine {
    std::vector<PhysicsObject*> objects;
    Vector2D gravity;
    void step(float dt);
}

abstract class Shape {
    virtual void draw() = 0;
}
class Circle : public Shape { float radius; }
class Rectangle : public Shape { float w, h; }

class PhysicsObject {
    Shape* shape;
    Vector2D pos, vel, acc;
    float mass;
    std::vector<Force*> forces;
    void applyForce(Force* f);
    void update(float dt);
}

This interface simulates the underlying C++ OOP architecture in TypeScript/JavaScript, demonstrating concepts like inheritance (Shape), polymorphism (draw methods), and vector mathematics.
